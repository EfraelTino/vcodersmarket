## Tipos de proyecto (Lanzamiento vs Directorio/Showcase)
create table public.project_types (
  id int generated by default as identity primary key,
  name text not null, -- 'Lanzamiento', 'Showcase'
  slug text unique not null
);

##  Etiquetas de Estado (En Venta, Busca CTO, etc.)
create table public.project_status_labels (
  id int generated by default as identity primary key,
  name text not null, -- 'En Venta', 'Busca Co-founder', 'Promoción', 'Beta Cerrada'
  color text, -- Para pintar el badge en el frontend (ej: 'bg-red-100')
  icon text   -- Nombre del icono
);

## Categorías (SaaS, DevTools, AI, etc.)
create table public.categories (
  id int generated by default as identity primary key,
  name text not null,
  slug text unique not null
);
# PORYECTOS
create table public.projects (
  id uuid default gen_random_uuid() primary key,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  user_id uuid references auth.users(id) not null, -- Dueño del proyecto
  type_id int references public.project_types(id),
  status_label_id int references public.project_status_labels(id),
  
  -- Info Básica
  name text not null,
  slug text unique not null, -- Para la url: vibemarket.com/p/mi-proyecto
  tagline text not null, -- Descripción corta (para la tarjeta)
  description text, -- Descripción larga (Markdown o HTML)
  
  -- Multimedia (Arrays de texto son más fáciles que tablas extra)
  logo_url text, 
  gallery_urls text[], -- Array de URLs de fotos: ['img1.png', 'img2.png']
  video_url text, -- Youtube URL
  
  -- Links Externos
  website_url text,
  repo_url text, -- Si es Open Source
  social_links jsonb, -- Ej: {"twitter": "url", "linkedin": "url", "discord": "url"}
  
  -- Métricas y Badges (Lo que pediste)
  stats_user_count int default 0, -- "1.250 usuarios"
  stats_mrr numeric default 0,    -- "$4.2k MRR"
  vibe_score int default 0,       -- "98%" (Tu métrica secreta)
  votes_count int default 0,      -- Total de votos (se actualiza con triggers o función)
  
  -- Flags (Booleanos para badges rápidos)
  is_open_source boolean default false,
  is_ai_powered boolean default false,
  is_verified boolean default false, -- ¿Tú verificaste que es real?
  
  -- Fechas
  launch_date timestamp with time zone
);
# cateogiras
create table public.project_categories (
  project_id uuid references public.projects(id) on delete cascade,
  category_id int references public.categories(id) on delete cascade,
  primary key (project_id, category_id)
);
